struct SearchConfig {
  double max_tau = 0.6, init_max_tau = 0.8, max_vel = 1.0, max_acc = 2.0;
  double max_vel_xy = 1.0, max_vel_z = 0.2, max_acc_xy = 2.0, max_acc_z = 2.0;
  double w_time = 10.0, horizon = 100.0, lambda_heu = 1.0;
  double vertical_distance_weight = 0.0;
  double resolution = 0.1, time_resolution = 0.8;
  double safe_distance = 0.25, voxel_size = 0.1, collision_step = 0.05;
  double search_budget = 0.3;
  int allocate_num = 100000, tree_period = 20;
  std::size_t max_cloud_points = 500000;
  Eigen::Vector3d lower{-50.0, -50.0, -2.0}, upper{50.0, 50.0, 10.0};
  void validate() const;
};
struct TrajectorySegment {
  double duration = 0.0;
  Eigen::Matrix<double, 3, 4> coefficients = Eigen::Matrix<double, 3, 4>::Zero();
};
struct TrajectorySample {
  double time = 0.0;
  Eigen::Vector3d position, velocity, acceleration;
};
enum NodeState { NOT_EXPAND, IN_OPEN_SET, IN_CLOSE_SET };
struct PathNode {
  Eigen::Vector3i index = Eigen::Vector3i::Zero();
  Eigen::Matrix<double, 6, 1> state = Eigen::Matrix<double, 6, 1>::Zero();
  double g_score = 0, f_score = 0, duration = 0, time = 0;
  Eigen::Vector3d input = Eigen::Vector3d::Zero();
  int time_idx = 0;
  PathNode* parent = nullptr;
  NodeState node_state = NOT_EXPAND;
};
using PathNodePtr = PathNode*;
struct QueueEntry { double score; PathNodePtr node; };
struct NodeComparator {
  bool operator()(const QueueEntry& a, const QueueEntry& b) const { return a.score > b.score; }
};
template <typename T> struct matrix_hash {
  std::size_t operator()(const T& m) const {
    std::size_t seed = 0;
    for (Eigen::Index i = 0; i < m.size(); ++i)
      seed ^= std::hash<typename T::Scalar>()(m.data()[i]) + 0x9e3779b9 + (seed << 6) + (seed >> 2);
    return seed;
  }
};
class NodeHashTable {
  std::unordered_map<Eigen::Vector3i, PathNodePtr, matrix_hash<Eigen::Vector3i>> data_3d_;
  std::unordered_map<Eigen::Vector4i, PathNodePtr, matrix_hash<Eigen::Vector4i>> data_4d_;
 public:
  void insert(Eigen::Vector3i i, PathNodePtr n) { data_3d_.emplace(i, n); }
  void insert(Eigen::Vector3i i, int t, PathNodePtr n) { data_4d_.emplace(Eigen::Vector4i(i.x(), i.y(), i.z(), t), n); }
  PathNodePtr find(Eigen::Vector3i i) { auto p = data_3d_.find(i); return p == data_3d_.end() ? nullptr : p->second; }
  PathNodePtr find(Eigen::Vector3i i, int t) {
    auto p = data_4d_.find(Eigen::Vector4i(i.x(), i.y(), i.z(), t));
    return p == data_4d_.end() ? nullptr : p->second;
  }
  void clear() { data_3d_.clear(); data_4d_.clear(); }
};
class KinodynamicAstar {
 public:
  enum { REACH_HORIZON = 1, REACH_END = 2, NO_PATH = 3, NEAR_END = 4,
         INVALID_INPUT = 5, NO_MAP = 6, TIMEOUT = 7, NODE_LIMIT = 8 };
  explicit KinodynamicAstar(const SearchConfig& config = SearchConfig());
  ~KinodynamicAstar();
  KinodynamicAstar(const KinodynamicAstar&) = delete;
  KinodynamicAstar& operator=(const KinodynamicAstar&) = delete;
  void reset();
  void clearMap();
  bool setKdtree(const pcl::PointCloud<pcl::PointXYZ>& cloud);
  bool mapReady() const { return map_ready_; }
  std::size_t mapPointCount() const;
  bool isSafe(double x, double y, double z);
  bool inBounds(const Eigen::Vector3d& p) const;
  int search(Eigen::Vector3d start, Eigen::Vector3d velocity, Eigen::Vector3d acceleration,
             Eigen::Vector3d goal, Eigen::Vector3d goal_velocity, bool init,
             bool dynamic = false, double time_start = 0.0);
  std::vector<TrajectorySegment> getSegments() const;
  static constexpr std::size_t kMaxTrajectorySamples = 1000000;
  std::vector<TrajectorySample> sampleTrajectory(double step) const;
  std::vector<TrajectorySample> sampleTrajectory(double step, std::size_t max_samples) const;
  std::vector<Eigen::Vector3d> getKinoTraj(double step) const;
  std::vector<PathNodePtr> getVisitedNodes();
  const SearchConfig& config() const { return config_; }
 private:
  SearchConfig config_;
  std::vector<PathNodePtr> path_node_pool_, path_nodes_;
  int use_node_num_ = 0, iter_num_ = 0;
  NodeHashTable expanded_nodes_;
  std::priority_queue<QueueEntry, std::vector<QueueEntry>, NodeComparator> open_set_;
  Eigen::Vector3d start_vel_, end_vel_, start_acc_;
  Eigen::Matrix<double, 6, 6> phi_ = Eigen::Matrix<double, 6, 6>::Identity();
  std::array<pcl::search::KdTree<pcl::PointXYZ>, 2> trees_;
  std::array<pcl::PointCloud<pcl::PointXYZ>::Ptr, 2> clouds_;
  std::vector<int> nearest_indices_{0};
  std::vector<float> nearest_distances_{0.0f};
  std::size_t cloud_input_num_ = 0;
  bool map_ready_ = false, is_shot_succ_ = false;
  Eigen::Matrix<double, 3, 4> coef_shot_ = Eigen::Matrix<double, 3, 4>::Zero();
  double t_shot_ = 0.0;
  double max_tau_, init_max_tau_, max_vel_, max_acc_, w_time_, horizon_, lambda_heu_;
  int allocate_num_;
  double tie_breaker_ = 1.0 + 1.0 / 10000.0;
  double resolution_, inv_resolution_, time_resolution_, inv_time_resolution_;
  Eigen::Vector3d origin_;
  double time_origin_ = 0.0;
  std::chrono::steady_clock::time_point deadline_;
  double primitiveClearance(const Eigen::Vector3d& position);
  bool isPrimitiveSafe(const Eigen::Matrix<double, 6, 1>& state, const Eigen::Vector3d& acceleration,
                       double begin, double end, double da, double db, int depth = 0);
  Eigen::Vector3i posToIndex(Eigen::Vector3d point);
  int timeToIndex(double time);
  void retrievePath(PathNodePtr node);
  std::vector<double> cubic(double a, double b, double c, double d);
  std::vector<double> quartic(double a, double b, double c, double d, double e);
  bool withinVelocity(const Eigen::Vector3d& value) const;
  bool withinAcceleration(const Eigen::Vector3d& value) const;
  bool computeShotTraj(Eigen::VectorXd start, Eigen::VectorXd end, double duration);
  double verticalTravel(const Eigen::Vector3d& coefficients, double duration);
  double shotCost();
  double estimateHeuristic(Eigen::VectorXd start, Eigen::VectorXd end, double& duration);
  void stateTransit(Eigen::Matrix<double, 6, 1>& start, Eigen::Matrix<double, 6, 1>& end,
                    Eigen::Vector3d acceleration, double duration);
};

constexpr double kMaxGridCellsPerAxis = 1e8;
constexpr double kMaxSearchBudgetSeconds = 60.0;
constexpr double kMaxPrimitiveCollisionChecks = 1e6;
constexpr double kNearGoalDistanceMeters = 1.0;
constexpr int kInitialDurationCount = 20;
constexpr double kMinHeuristicTimeSeconds = 1e-3;
constexpr double kHeuristicVelocityFraction = 0.5;
constexpr double kInitialHeuristicCost = 100000000.0;
constexpr double kAccelerationZeroTolerance = 1e-12;
constexpr double kPolynomialZeroTolerance = 1e-12;
constexpr double kDerivativeLimitSlack = 1e-9;
constexpr std::array<double, 5> kShotDurationScales{1.0, 1.5, 2.0, 3.0, 4.0};
constexpr int kPrimitiveSubdivisionLimit = 14;
constexpr double kNearestDistanceSlack = 8 * std::numeric_limits<float>::epsilon();
}
void SearchConfig::validate() const {
  if (!std::isfinite(vertical_distance_weight) || vertical_distance_weight < 0)
    throw std::invalid_argument("vertical_distance_weight must be finite and nonnegative");
  for (double v : {max_tau, init_max_tau, max_vel, max_acc, max_vel_xy, max_vel_z, max_acc_xy, max_acc_z, w_time, horizon, lambda_heu,
                   resolution, time_resolution, safe_distance, voxel_size, collision_step, search_budget})
    if (!std::isfinite(v) || v <= 0) throw std::invalid_argument("search parameters must be finite and positive");
  const double min_tau = init_max_tau * (1.0 / kInitialDurationCount);
  if (max_acc / 2 == 0 || min_tau == 0 || min_tau * min_tau == 0 || max_tau * max_tau == 0 ||
      !std::isfinite(max_tau * max_tau) || !std::isfinite(init_max_tau * init_max_tau) ||
      !std::isfinite(3 * max_acc * max_acc) || !std::isfinite(1 / resolution) ||
      !std::isfinite(1 / time_resolution) ||
      !std::isfinite(max_acc * std::max(max_tau * max_tau, init_max_tau * init_max_tau)) ||
      !std::isfinite(max_vel * std::max(max_tau, init_max_tau)))
    throw std::invalid_argument("search parameters exceed representable acceleration, duration or index range");
  if (allocate_num < 2 || tree_period < 1 || max_cloud_points < 1 || !lower.allFinite() || !upper.allFinite() ||
      (lower.array() >= upper.array()).any() || ((upper-lower).array()/resolution > kMaxGridCellsPerAxis).any() ||
      collision_step > safe_distance || search_budget > kMaxSearchBudgetSeconds ||
      std::max(max_tau, init_max_tau)*std::sqrt(3.0)*max_vel/collision_step > kMaxPrimitiveCollisionChecks)
    throw std::invalid_argument("invalid map bounds, capacities, sampling step or search budget");
}
KinodynamicAstar::KinodynamicAstar(const SearchConfig& c) : config_(c) {
  c.validate();
  max_tau_=c.max_tau; init_max_tau_=c.init_max_tau; max_vel_=c.max_vel; max_acc_=c.max_acc;
  w_time_=c.w_time; horizon_=c.horizon; lambda_heu_=c.lambda_heu; allocate_num_=c.allocate_num;
  resolution_=c.resolution; inv_resolution_=1/c.resolution;
  time_resolution_=c.time_resolution; inv_time_resolution_=1/c.time_resolution; origin_=c.lower;
  path_node_pool_.reserve(allocate_num_);
  for (int i=0; i<allocate_num_; ++i) path_node_pool_.push_back(new PathNode);
  clearMap();
}
KinodynamicAstar::~KinodynamicAstar() { for (auto node : path_node_pool_) delete node; }
void KinodynamicAstar::reset() {
  expanded_nodes_.clear(); path_nodes_.clear(); open_set_ = decltype(open_set_){};
  for (int i=0; i<use_node_num_; ++i) *path_node_pool_[i] = PathNode{};
  use_node_num_=0; iter_num_=0; is_shot_succ_=false; t_shot_=0;
}
void KinodynamicAstar::clearMap() {
  for (std::size_t i=0; i<2; ++i) {
    trees_[i] = pcl::search::KdTree<pcl::PointXYZ>();
    clouds_[i].reset(new pcl::PointCloud<pcl::PointXYZ>);
  }
  cloud_input_num_=0; map_ready_=false;
  reset();
}
std::size_t KinodynamicAstar::mapPointCount() const { return clouds_[0]->size()+clouds_[1]->size(); }
bool KinodynamicAstar::setKdtree(const pcl::PointCloud<pcl::PointXYZ>& input) {
  if (input.empty() || input.size() > config_.max_cloud_points) { clearMap(); return false; }
  for (const auto& p : input)
    if (!std::isfinite(p.x) || !std::isfinite(p.y) || !std::isfinite(p.z)) { clearMap(); return false; }
  const auto bank = (cloud_input_num_/config_.tree_period)%2;
  auto next = std::make_shared<pcl::PointCloud<pcl::PointXYZ>>();
  if (cloud_input_num_%config_.tree_period != 0) *next = *clouds_[bank];
  *next += input;
  pcl::VoxelGrid<pcl::PointXYZ> filter;
  filter.setInputCloud(next);
  filter.setLeafSize(config_.voxel_size, config_.voxel_size, config_.voxel_size);
  auto filtered = std::make_shared<pcl::PointCloud<pcl::PointXYZ>>();
  filter.filter(*filtered);
  if (filtered->empty() || filtered->size()+clouds_[1-bank]->size() > config_.max_cloud_points) {
    clearMap(); return false;
  }
  clouds_[bank] = filtered; trees_[bank].setInputCloud(filtered);
  ++cloud_input_num_; map_ready_=true;
  return true;
}
bool KinodynamicAstar::inBounds(const Eigen::Vector3d& p) const {
  return p.allFinite() && (p.array() >= config_.lower.array()).all() && (p.array() <= config_.upper.array()).all();
}
bool KinodynamicAstar::isSafe(double x, double y, double z) {
  if (!map_ready_ || !inBounds({x,y,z})) return false;
  const double radius = config_.safe_distance;
  for (std::size_t i=0; i<2; ++i) {
    if (clouds_[i]->empty()) continue;
    if (trees_[i].nearestKSearch(pcl::PointXYZ(x,y,z), 1, nearest_indices_, nearest_distances_) != 1 ||
        nearest_distances_[0] < radius*radius) return false;
  }
  return true;
}
double KinodynamicAstar::primitiveClearance(const Eigen::Vector3d& position) {
  if (!map_ready_ || !inBounds(position)) return 0;
  const pcl::PointXYZ query(position.x(), position.y(), position.z());
  const double query_error = (position - Eigen::Vector3d(query.x, query.y, query.z)).norm();
  double distance = std::numeric_limits<double>::infinity();
  for (std::size_t bank = 0; bank < 2; ++bank) {
    if (clouds_[bank]->empty()) continue;
    if (trees_[bank].nearestKSearch(query, 1, nearest_indices_, nearest_distances_) != 1) return 0;
    distance = std::min(distance, std::sqrt(static_cast<double>(nearest_distances_[0])) *
                                 (1 - kNearestDistanceSlack) - query_error);
    if (distance < config_.safe_distance) break;
  }
  return distance;
}
bool KinodynamicAstar::isPrimitiveSafe(const Eigen::Matrix<double, 6, 1>& state,
    const Eigen::Vector3d& acceleration, double begin, double end, double da, double db, int depth) {
  if (std::chrono::steady_clock::now() >= deadline_) return false;
  if (da < config_.safe_distance || db < config_.safe_distance) return false;
  const auto velocity = state.tail(3);
  const double length_bound = (end - begin) * 0.5 * ((velocity + acceleration * begin).norm() +
                                                   (velocity + acceleration * end).norm());
  if (da + db >= 2 * config_.safe_distance + length_bound) return true;
  if (depth >= kPrimitiveSubdivisionLimit) return false;
  const double mid = (begin + end) * 0.5;
  const Eigen::Vector3d position = state.head(3) + velocity * mid + 0.5 * acceleration * mid * mid;
  const double dm = primitiveClearance(position);
  if (dm < config_.safe_distance) return false;
  return isPrimitiveSafe(state, acceleration, begin, mid, da, dm, depth + 1) &&
         isPrimitiveSafe(state, acceleration, mid, end, dm, db, depth + 1);
}
int KinodynamicAstar::search(Eigen::Vector3d start_pt, Eigen::Vector3d start_v, Eigen::Vector3d start_a,
                             Eigen::Vector3d end_pt, Eigen::Vector3d end_v, bool init, bool dynamic, double time_start)
{
  reset();
  deadline_ = std::chrono::steady_clock::now() + std::chrono::duration_cast<std::chrono::steady_clock::duration>(
      std::chrono::duration<double>(config_.search_budget));
  if (!start_pt.allFinite() || !start_v.allFinite() || !start_a.allFinite() ||
      !end_pt.allFinite() || !end_v.allFinite() || !std::isfinite(time_start) ||
      !inBounds(start_pt) || !inBounds(end_pt) ||
      end_v.cwiseAbs().maxCoeff()>max_vel_ ||
      start_a.cwiseAbs().maxCoeff()>max_acc_ ||
      !withinVelocity(end_v) || !withinAcceleration(start_a)) return INVALID_INPUT;
  if (!map_ready_) return NO_MAP;
  if (!isSafe(start_pt.x(), start_pt.y(), start_pt.z()) ||
      !isSafe(end_pt.x(), end_pt.y(), end_pt.z())) return NO_PATH;
  time_origin_ = time_start;
  start_vel_ = start_v;
  start_acc_ = start_a;
  PathNodePtr cur_node = path_node_pool_[0];
  *cur_node = PathNode{};
  cur_node->time = time_start;
  cur_node->parent = NULL;
  cur_node->state.head(3) = start_pt;
  cur_node->state.tail(3) = start_v;
  cur_node->index = posToIndex(start_pt);
  cur_node->g_score = 0.0;
  Eigen::VectorXd end_state(6);
  Eigen::Vector3i end_index;
  double time_to_goal;
  end_state.head(3) = end_pt;
  end_state.tail(3) = end_v;
  end_index = posToIndex(end_pt);
  cur_node->f_score = lambda_heu_ * estimateHeuristic(cur_node->state, end_state, time_to_goal);
  cur_node->node_state = IN_OPEN_SET;
  open_set_.push({cur_node->f_score, cur_node});
  use_node_num_ += 1;
  if (dynamic)
  {
    time_origin_ = time_start;
    cur_node->time = time_start;
    cur_node->time_idx = timeToIndex(time_start);
    expanded_nodes_.insert(cur_node->index, cur_node->time_idx, cur_node);
  }
  else
    expanded_nodes_.insert(cur_node->index, cur_node);
  PathNodePtr terminate_node = NULL;
  PathNodePtr best_terminal = nullptr;
  Eigen::Matrix<double, 3, 4> best_shot;
  double best_duration = 0.0, best_cost = std::numeric_limits<double>::infinity();
  const bool prefer_horizontal = config_.vertical_distance_weight > 0;
  const auto finish = [&](int failure) {
    if (best_terminal) {
      retrievePath(best_terminal);
      coef_shot_ = best_shot; t_shot_ = best_duration; is_shot_succ_ = true;
      return int(REACH_END);
    }
    reset();
    return failure;
  };
  bool init_search = init;
  const int tolerance = ceil(kNearGoalDistanceMeters / resolution_);
  while (!open_set_.empty())
  {
    if (std::chrono::steady_clock::now() >= deadline_) return finish(TIMEOUT);
    const auto entry = open_set_.top();
    cur_node = entry.node;
    if (cur_node->node_state != IN_OPEN_SET || entry.score != cur_node->f_score) {
      open_set_.pop();
      continue;
    }
    if (best_terminal && entry.score >= best_cost) return finish(REACH_END);
    bool reach_horizon = (cur_node->state.head(3) - start_pt).norm() >= horizon_;
    bool near_end = withinVelocity(cur_node->state.tail(3)) &&
                    abs(cur_node->index(0) - end_index(0)) <= tolerance &&
                    abs(cur_node->index(1) - end_index(1)) <= tolerance &&
                    abs(cur_node->index(2) - end_index(2)) <= tolerance;
    if (reach_horizon || near_end)
    {
      terminate_node = cur_node;
      retrievePath(terminate_node);
      if (near_end)
      {
        estimateHeuristic(cur_node->state, end_state, time_to_goal);
        is_shot_succ_ = false;
        for (double scale : kShotDurationScales) {
          if (computeShotTraj(cur_node->state, end_state, time_to_goal * scale)) {
            if (!prefer_horizontal) break;
            const double cost = cur_node->g_score + shotCost();
            if (cost < best_cost) {
              best_terminal = cur_node; best_shot = coef_shot_;
              best_duration = t_shot_; best_cost = cost;
            }
          }
          if (std::chrono::steady_clock::now() >= deadline_) break;
        }
        if (std::chrono::steady_clock::now() >= deadline_) return finish(TIMEOUT);
      }
    }
    if (reach_horizon)
    {
      if (best_terminal) return finish(REACH_END);
      if (is_shot_succ_ && !prefer_horizontal)
      {
        return REACH_END;
      }
      else
      {
        return REACH_HORIZON;
      }
    }
    if (near_end)
    {
      if (is_shot_succ_ && !prefer_horizontal)
      {
        return REACH_END;
      }
    }
    path_nodes_.clear();
    is_shot_succ_ = false;
    open_set_.pop();
    cur_node->node_state = IN_CLOSE_SET;
    iter_num_ += 1;
    Eigen::Matrix<double, 6, 1> cur_state = cur_node->state;
    const double primitive_start_distance = primitiveClearance(cur_state.head(3));
    Eigen::Matrix<double, 6, 1> pro_state;
    vector<PathNodePtr> tmp_expand_nodes;
    Eigen::Vector3d um;
    double pro_t;
    vector<Eigen::Vector3d> inputs;
    vector<double> durations;
    inputs.reserve(init_search ? 1 : 150);
    durations.reserve(init_search ? kInitialDurationCount : 1);
    const bool recovering_root=cur_node->parent==nullptr && !withinVelocity(cur_state.tail(3));
    const bool initial_expansion=init_search || recovering_root;
    if (recovering_root)
    {
      Eigen::Vector3d recovered=cur_state.tail(3).cwiseMax(-max_vel_).cwiseMin(max_vel_);
      if (recovered.head<2>().norm()>config_.max_vel_xy)
        recovered.head<2>()*=config_.max_vel_xy/recovered.head<2>().norm();
      recovered.z()=std::clamp(recovered.z(),-config_.max_vel_z,config_.max_vel_z);
      const Eigen::Vector3d change=recovered-cur_state.tail(3);
      const double duration=std::max({change.head<2>().norm()/config_.max_acc_xy,
          std::abs(change.z())/config_.max_acc_z,change.cwiseAbs().maxCoeff()/max_acc_});
      if (!std::isfinite(duration) || duration<=0) return finish(INVALID_INPUT);
      inputs.push_back(change/duration);
      durations.push_back(duration);
      init_search=false;
    }
    else if (init_search)
    {
      inputs.push_back(start_acc_);
      for (int i = 1; i <= kInitialDurationCount; ++i)
        durations.push_back(init_max_tau_ * (double(i) / kInitialDurationCount));
      init_search = false;
    }
    else
    {
      std::array<std::array<double, 6>, 3> axes;
      std::array<std::size_t, 3> counts{5,5,5};
      for (int axis=0; axis<3; ++axis) {
        const double acceleration=std::min(max_acc_,axis==2?config_.max_acc_z:config_.max_acc_xy);
        const double velocity=std::min(max_vel_,axis==2?config_.max_vel_z:config_.max_vel_xy);
        axes[axis]={-acceleration,-acceleration/2,0,acceleration/2,acceleration};
        if (acceleration*max_tau_/2<=velocity) continue;
        const double negative=std::min(acceleration,std::max(0.0,(velocity+cur_state(axis+3))/max_tau_));
        const double positive=std::min(acceleration,std::max(0.0,(velocity-cur_state(axis+3))/max_tau_));
        axes[axis]={-negative,-negative/2,0,positive/2,positive};
        const double stop=-cur_state(axis+3)/max_tau_;
        if (axis==2 && std::abs(stop)<=acceleration &&
            std::none_of(axes[axis].begin(),axes[axis].begin()+counts[axis],
                [&](double value) { return std::abs(value-stop)<kAccelerationZeroTolerance; })) {
          axes[axis][counts[axis]++]=stop;
        }
      }
      for (std::size_t ix=0; ix<counts[0]; ++ix)
        for (std::size_t iy=0; iy<counts[1]; ++iy)
          for (std::size_t iz=0; iz<counts[2]; ++iz)
          {
            um << axes[0][ix], axes[1][iy], axes[2][iz];
            inputs.push_back(um);
          }
      durations.push_back(max_tau_);
    }
    for (std::size_t i = 0; i < inputs.size(); ++i)
      for (std::size_t j = 0; j < durations.size(); ++j)
      {
        um = inputs[i];
        if (std::chrono::steady_clock::now() >= deadline_) return finish(TIMEOUT);
        if (!um.allFinite() || !withinAcceleration(um)) continue;
        double tau = durations[j];
        stateTransit(cur_state, pro_state, um, tau);
        pro_t = cur_node->time + tau;
        if (!pro_state.allFinite()) continue;
        if (dynamic && (!std::isfinite(pro_t) || pro_t <= cur_node->time ||
            !std::isfinite((pro_t - time_origin_) * inv_time_resolution_) ||
            (pro_t - time_origin_) * inv_time_resolution_ > std::numeric_limits<int>::max())) {
          reset(); return INVALID_INPUT;
        }
        Eigen::Vector3d pro_pos = pro_state.head(3);
        if (!inBounds(pro_pos)) continue;
        bool leaves_bounds = false;
        for (int axis=0; axis<3; ++axis) {
          if (std::abs(um(axis)) < kAccelerationZeroTolerance) continue;
          const double t = -cur_state(axis+3)/um(axis);
          if (t>0 && t<tau) {
            const double p = cur_state(axis)+cur_state(axis+3)*t+0.5*um(axis)*t*t;
            leaves_bounds |= p<config_.lower(axis) || p>config_.upper(axis);
          }
        }
        if (leaves_bounds) continue;
        Eigen::Vector3i pro_id = posToIndex(pro_pos);
        int pro_t_id = dynamic ? timeToIndex(pro_t) : 0;
        const bool initial_same_cell=initial_expansion && !dynamic && pro_id==cur_node->index;
        PathNodePtr pro_node = initial_same_cell ? nullptr :
            (dynamic ? expanded_nodes_.find(pro_id, pro_t_id) : expanded_nodes_.find(pro_id));
        if (pro_node != NULL && pro_node->node_state == IN_CLOSE_SET)
        {
            continue;
        }
        Eigen::Vector3d pro_v = pro_state.tail(3);
        if (!withinVelocity(pro_v))
        {
            continue;
        }
        Eigen::Vector3i diff = pro_id - cur_node->index;
        int diff_time = pro_t_id - cur_node->time_idx;
        if (!initial_same_cell && diff.norm() == 0 && ((!dynamic) || diff_time == 0))
        {
            continue;
        }
        if (!isPrimitiveSafe(cur_state, um, 0, tau, primitive_start_distance,
                             primitiveClearance(pro_pos)))
        {
            if (std::chrono::steady_clock::now() >= deadline_) return finish(TIMEOUT);
            continue;
        }
        double time_to_goal, tmp_g_score, tmp_f_score;
        tmp_g_score = (um.squaredNorm() + w_time_) * tau + cur_node->g_score;
        if (prefer_horizontal)
          tmp_g_score += config_.vertical_distance_weight *
              verticalTravel({cur_state(5), 0.5 * um.z(), 0.0}, tau);
        tmp_f_score = tmp_g_score + lambda_heu_ * estimateHeuristic(pro_state, end_state, time_to_goal);
        bool prune = false;
        for (std::size_t j = 0; j < tmp_expand_nodes.size(); ++j)
        {
          PathNodePtr expand_node = tmp_expand_nodes[j];
          if ((pro_id - expand_node->index).norm() == 0 && ((!dynamic) || pro_t_id == expand_node->time_idx))
          {
            prune = true;
            if (tmp_f_score < expand_node->f_score)
            {
              expand_node->f_score = tmp_f_score;
              expand_node->g_score = tmp_g_score;
              expand_node->state = pro_state;
              expand_node->input = um;
              expand_node->duration = tau;
              open_set_.push({expand_node->f_score, expand_node});
              if (dynamic)
                expand_node->time = cur_node->time + tau;
            }
            break;
          }
        }
        if (!prune)
        {
          if (pro_node == NULL)
          {
            if (use_node_num_ >= allocate_num_) return finish(NODE_LIMIT);
            pro_node = path_node_pool_[use_node_num_];
            pro_node->index = pro_id;
            pro_node->state = pro_state;
            pro_node->f_score = tmp_f_score;
            pro_node->g_score = tmp_g_score;
            pro_node->input = um;
            pro_node->duration = tau;
            pro_node->parent = cur_node;
            pro_node->node_state = IN_OPEN_SET;
            if (dynamic)
            {
              pro_node->time = cur_node->time + tau;
              pro_node->time_idx = timeToIndex(pro_node->time);
            }
            open_set_.push({pro_node->f_score, pro_node});
            if (dynamic)
              expanded_nodes_.insert(pro_id, pro_node->time_idx, pro_node);
            else if (!initial_same_cell)
              expanded_nodes_.insert(pro_id, pro_node);
            if (!initial_same_cell) tmp_expand_nodes.push_back(pro_node);
            use_node_num_ += 1;
          }
          else if (pro_node->node_state == IN_OPEN_SET)
          {
            if (tmp_g_score < pro_node->g_score)
            {
              pro_node->state = pro_state;
              pro_node->f_score = tmp_f_score;
              pro_node->g_score = tmp_g_score;
              pro_node->input = um;
              pro_node->duration = tau;
              pro_node->parent = cur_node;
              open_set_.push({pro_node->f_score, pro_node});
              if (dynamic)
                pro_node->time = cur_node->time + tau;
            }
          }
        }
      }
  }
  return finish(NO_PATH);
}
void KinodynamicAstar::retrievePath(PathNodePtr end_node)
{
  PathNodePtr cur_node = end_node;
  path_nodes_.clear();
  path_nodes_.push_back(cur_node);
  while (cur_node->parent != NULL)
  {
    cur_node = cur_node->parent;
    path_nodes_.push_back(cur_node);
  }
  reverse(path_nodes_.begin(), path_nodes_.end());
}
double KinodynamicAstar::estimateHeuristic(Eigen::VectorXd x1, Eigen::VectorXd x2, double& optimal_time)
{
  const Vector3d dp = x2.head(3) - x1.head(3);
  const Vector3d v0 = x1.segment(3, 3);
  const Vector3d v1 = x2.segment(3, 3);
  double c1 = -36 * dp.dot(dp);
  double c2 = 24 * (v0 + v1).dot(dp);
  double c3 = -4 * (v0.dot(v0) + v0.dot(v1) + v1.dot(v1));
  double c4 = 0;
  double c5 = w_time_;
  std::vector<double> ts = quartic(c5, c4, c3, c2, c1);
  const double horizontal=std::min(max_vel_,config_.max_vel_xy)*kHeuristicVelocityFraction;
  const double vertical=std::min(max_vel_,config_.max_vel_z)*kHeuristicVelocityFraction;
  double t_bar = std::max({kMinHeuristicTimeSeconds, dp.head<2>().norm()/horizontal, std::abs(dp.z())/vertical});
  ts.push_back(t_bar);
  double cost = kInitialHeuristicCost;
  double t_d = t_bar;
  for (auto t : ts)
  {
    if (!std::isfinite(t) || t < t_bar)
      continue;
    double c = -c1 / (3 * t * t * t) - c2 / (2 * t * t) - c3 / t + w_time_ * t;
    if (c < cost)
    {
      cost = c;
      t_d = t;
    }
  }
  optimal_time = t_d;
  double heuristic = tie_breaker_ * cost;
  return heuristic;
}
double KinodynamicAstar::verticalTravel(const Eigen::Vector3d& c, double duration) {
  const auto displacement = [&](double t) { return ((c.z() * t + c.y()) * t + c.x()) * t; };
  if (c.z() == 0.0) {
    const double end = displacement(duration);
    const double stop = c.y() != 0.0 ? -c.x() / (2 * c.y()) : -1.0;
    if (stop > 0.0 && stop < duration) {
      const double turn = displacement(stop);
      return std::abs(turn) + std::abs(end - turn);
    }
    return std::abs(end);
  }
  auto turns = cubic(0.0, 3 * c.z() * duration * duration, 2 * c.y() * duration, c.x());
  turns.push_back(1.0);
  std::sort(turns.begin(), turns.end());
  double travel = 0.0, previous = 0.0;
  for (double u : turns) {
    if (u <= 0.0 || u > 1.0) continue;
    const double position = displacement(u * duration);
    travel += std::abs(position - previous);
    previous = position;
  }
  return travel;
}
double KinodynamicAstar::shotCost() {
  const Eigen::Vector3d first = 2 * coef_shot_.col(2);
  const Eigen::Vector3d last = first + 6 * coef_shot_.col(3) * t_shot_;
  return t_shot_ * ((first.squaredNorm() + first.dot(last) + last.squaredNorm()) / 3 + w_time_) +
      config_.vertical_distance_weight * verticalTravel(coef_shot_.row(2).tail<3>().transpose(), t_shot_);
}
bool KinodynamicAstar::withinVelocity(const Eigen::Vector3d& value) const {
  return value.cwiseAbs().maxCoeff()<=max_vel_+kDerivativeLimitSlack &&
      value.head<2>().norm()<=config_.max_vel_xy+kDerivativeLimitSlack &&
      std::abs(value.z())<=config_.max_vel_z+kDerivativeLimitSlack;
}
bool KinodynamicAstar::withinAcceleration(const Eigen::Vector3d& value) const {
  return value.cwiseAbs().maxCoeff()<=max_acc_+kDerivativeLimitSlack &&
      value.head<2>().norm()<=config_.max_acc_xy+kDerivativeLimitSlack &&
      std::abs(value.z())<=config_.max_acc_z+kDerivativeLimitSlack;
}
bool KinodynamicAstar::computeShotTraj(Eigen::VectorXd state1, Eigen::VectorXd state2, double time_to_goal)
{
  const Vector3d p0 = state1.head(3);
  const Vector3d dp = state2.head(3) - p0;
  const Vector3d v0 = state1.segment(3, 3);
  const Vector3d v1 = state2.segment(3, 3);
  const Vector3d dv = v1 - v0;
  double t_d = time_to_goal;
  if (!std::isfinite(t_d) || t_d <= 0) return false;
  MatrixXd coef(3, 4);
  end_vel_ = v1;
  Vector3d a = 1.0 / 6.0 * (-12.0 / (t_d * t_d * t_d) * (dp - v0 * t_d) + 6 / (t_d * t_d) * dv);
  Vector3d b = 0.5 * (6.0 / (t_d * t_d) * (dp - v0 * t_d) - 2 / t_d * dv);
  Vector3d c = v0;
  Vector3d d = p0;
  coef.col(3) = a, coef.col(2) = b, coef.col(1) = c, coef.col(0) = d;
  if (!coef.allFinite()) return false;
  if (!withinAcceleration(2*b) || !withinAcceleration(2*b+6*a*t_d)) return false;
  const Eigen::Vector2d q0=c.head<2>(), q1=2*b.head<2>()*t_d, q2=3*a.head<2>()*t_d*t_d;
  auto speed_times=cubic(2*q2.squaredNorm(), 3*q1.dot(q2),
                         q1.squaredNorm()+2*q0.dot(q2), q0.dot(q1));
  speed_times.push_back(0); speed_times.push_back(1);
  for (double u:speed_times) {
    if (u>=0 && u<=1 && !withinVelocity(c+2*b*(u*t_d)+3*a*(u*t_d)*(u*t_d))) return false;
  }
  for (int axis = 0; axis < 3; ++axis) {
    const auto position = [&](double t) { return d(axis) + c(axis)*t + b(axis)*t*t + a(axis)*t*t*t; };
    const auto velocity = [&](double t) { return c(axis) + 2*b(axis)*t + 3*a(axis)*t*t; };
    if (std::max(std::abs(2*b(axis)), std::abs(2*b(axis)+6*a(axis)*t_d)) > max_acc_ + kDerivativeLimitSlack)
      return false;
    std::vector<double> velocity_times{0.0, t_d};
    if (std::abs(a(axis)) > kPolynomialZeroTolerance) velocity_times.push_back(-b(axis)/(3*a(axis)));
    for (double t : velocity_times)
      if (t >= 0 && t <= t_d && std::abs(velocity(t)) > std::min(max_vel_,axis==2?config_.max_vel_z:config_.max_vel_xy) + kDerivativeLimitSlack) return false;
    std::vector<double> position_times{0.0, t_d};
    if (std::abs(a(axis)) > kPolynomialZeroTolerance) {
      const double disc = 4*b(axis)*b(axis)-12*a(axis)*c(axis);
      if (disc >= 0) {
        position_times.push_back((-2*b(axis)+std::sqrt(disc))/(6*a(axis)));
        position_times.push_back((-2*b(axis)-std::sqrt(disc))/(6*a(axis)));
      }
    } else if (std::abs(b(axis)) > kPolynomialZeroTolerance) position_times.push_back(-c(axis)/(2*b(axis)));
    for (double t : position_times)
      if (t >= 0 && t <= t_d && (position(t) < config_.lower(axis) || position(t) > config_.upper(axis)))
        return false;
  }
  const double check_count = std::ceil(t_d * std::hypot(config_.max_vel_xy,config_.max_vel_z) / config_.collision_step);
  if (!coef.allFinite() || !std::isfinite(check_count) || check_count >= std::numeric_limits<int>::max())
    return false;
  const int checks = std::max(1, static_cast<int>(check_count));
  for (int i = 0; i <= checks; ++i) {
    if (std::chrono::steady_clock::now() >= deadline_) return false;
    const double t = t_d * i / checks;
    const Vector3d p = d + c*t + b*t*t + a*t*t*t;
    if (!isSafe(p.x(), p.y(), p.z())) return false;
  }
  coef_shot_ = coef;
  t_shot_ = t_d;
  is_shot_succ_ = true;
  return true;
}
vector<double> KinodynamicAstar::cubic(double a, double b, double c, double d)
{
  vector<double> dts;
  if (std::abs(a)<kPolynomialZeroTolerance) {
    if (std::abs(b)<kPolynomialZeroTolerance) {
      if (std::abs(c)>=kPolynomialZeroTolerance) dts.push_back(-d/c);
    } else {
      const double discriminant=c*c-4*b*d;
      if (discriminant>=0) {
        dts.push_back((-c+std::sqrt(discriminant))/(2*b));
        dts.push_back((-c-std::sqrt(discriminant))/(2*b));
      }
    }
    return dts;
  }
  double a2 = b / a;
  double a1 = c / a;
  double a0 = d / a;
  double Q = (3 * a1 - a2 * a2) / 9;
  double R = (9 * a1 * a2 - 27 * a0 - 2 * a2 * a2 * a2) / 54;
  double D = Q * Q * Q + R * R;
  if (D > 0)
  {
    double S = std::cbrt(R + sqrt(D));
    double T = std::cbrt(R - sqrt(D));
    dts.push_back(-a2 / 3 + (S + T));
    return dts;
  }
  else if (D == 0)
  {
    double S = std::cbrt(R);
    dts.push_back(-a2 / 3 + S + S);
    dts.push_back(-a2 / 3 - S);
    return dts;
  }
  else
  {
    double theta = acos(std::clamp(R / sqrt(-Q * Q * Q), -1.0, 1.0));
    dts.push_back(2 * sqrt(-Q) * cos(theta / 3) - a2 / 3);
    dts.push_back(2 * sqrt(-Q) * cos((theta + 2 * M_PI) / 3) - a2 / 3);
    dts.push_back(2 * sqrt(-Q) * cos((theta + 4 * M_PI) / 3) - a2 / 3);
    return dts;
  }
}
vector<double> KinodynamicAstar::quartic(double a, double b, double c, double d, double e)
{
  vector<double> dts;
  double a3 = b / a;
  double a2 = c / a;
  double a1 = d / a;
  double a0 = e / a;
  vector<double> ys = cubic(1, -a2, a1 * a3 - 4 * a0, 4 * a2 * a0 - a1 * a1 - a3 * a3 * a0);
  double y1 = ys.front();
  double r = a3 * a3 / 4 - a2 + y1;
  if (r < 0)
    return dts;
  double R = sqrt(r);
  double D, E;
  if (R != 0)
  {
    D = sqrt(0.75 * a3 * a3 - R * R - 2 * a2 + 0.25 * (4 * a3 * a2 - 8 * a1 - a3 * a3 * a3) / R);
    E = sqrt(0.75 * a3 * a3 - R * R - 2 * a2 - 0.25 * (4 * a3 * a2 - 8 * a1 - a3 * a3 * a3) / R);
  }
  else
  {
    D = sqrt(0.75 * a3 * a3 - 2 * a2 + 2 * sqrt(y1 * y1 - 4 * a0));
    E = sqrt(0.75 * a3 * a3 - 2 * a2 - 2 * sqrt(y1 * y1 - 4 * a0));
  }
  if (!std::isnan(D))
  {
    dts.push_back(-a3 / 4 + R / 2 + D / 2);
    dts.push_back(-a3 / 4 + R / 2 - D / 2);
  }
  if (!std::isnan(E))
  {
    dts.push_back(-a3 / 4 - R / 2 + E / 2);
    dts.push_back(-a3 / 4 - R / 2 - E / 2);
  }
  return dts;
}
Eigen::Vector3i KinodynamicAstar::posToIndex(Eigen::Vector3d pt)
{
  Vector3i idx = ((pt - origin_) * inv_resolution_).array().floor().cast<int>();
  return idx;
}
int KinodynamicAstar::timeToIndex(double time)
{
  return static_cast<int>(std::floor((time - time_origin_) * inv_time_resolution_));
}
void KinodynamicAstar::stateTransit(Eigen::Matrix<double, 6, 1>& state0, Eigen::Matrix<double, 6, 1>& state1,
                                    Eigen::Vector3d um, double tau)
{
  for (int i = 0; i < 3; ++i)
    phi_(i, i + 3) = tau;
  Eigen::Matrix<double, 6, 1> integral;
  integral.head(3) = 0.5 * pow(tau, 2) * um;
  integral.tail(3) = tau * um;
  state1 = phi_ * state0 + integral;
}
std::vector<TrajectorySegment> KinodynamicAstar::getSegments() const {
  std::vector<TrajectorySegment> out;
  for (const auto node : path_nodes_) {
    if (!node->parent) continue;
    TrajectorySegment s;
    s.duration = node->duration;
    s.coefficients.col(0) = node->parent->state.head<3>();
    s.coefficients.col(1) = node->parent->state.tail<3>();
    s.coefficients.col(2) = node->input/2;
    out.push_back(s);
  }
  if (is_shot_succ_) out.push_back({t_shot_, coef_shot_});
  return out;
}
std::vector<TrajectorySample> KinodynamicAstar::sampleTrajectory(double step) const {
  return sampleTrajectory(step, kMaxTrajectorySamples);
}
std::vector<TrajectorySample> KinodynamicAstar::sampleTrajectory(double step, std::size_t max_samples) const {
  if (!std::isfinite(step) || step <= 0) throw std::invalid_argument("sample step must be positive");
  const auto segments = getSegments();
  const auto limit = std::min(max_samples, kMaxTrajectorySamples);
  std::size_t sample_count = 0;
  for (const auto& s : segments) {
    const double count = std::ceil(s.duration / step);
    if (!std::isfinite(count) || count > limit)
      throw std::length_error("too many trajectory samples");
    const auto n = std::max<std::size_t>(1, static_cast<std::size_t>(count));
    const auto added = n + (sample_count == 0 ? 1 : 0);
    if (added > limit - sample_count) throw std::length_error("too many trajectory samples");
    sample_count += added;
  }
  std::vector<TrajectorySample> out;
  out.reserve(sample_count);
  double elapsed = 0;
  for (const auto& s : segments) {
    const double count = std::ceil(s.duration/step);
    const int n = std::max(1, static_cast<int>(count));
    for (int i = out.empty() ? 0 : 1; i <= n; ++i) {
      const double t = s.duration*i/n;
      const auto& c = s.coefficients;
      out.push_back({elapsed+t, c.col(0)+c.col(1)*t+c.col(2)*t*t+c.col(3)*t*t*t,
          c.col(1)+2*c.col(2)*t+3*c.col(3)*t*t, 2*c.col(2)+6*c.col(3)*t});
    }
    elapsed += s.duration;
  }
  return out;
}
std::vector<Eigen::Vector3d> KinodynamicAstar::getKinoTraj(double step) const {
  std::vector<Eigen::Vector3d> out;
  for (const auto& sample : sampleTrajectory(step)) out.push_back(sample.position);
  return out;
}
std::vector<PathNodePtr> KinodynamicAstar::getVisitedNodes() {
  return {path_node_pool_.begin(), path_node_pool_.begin()+use_node_num_};
}
#pragma once
#include <path_searching/kinodynamic_astar.h>
#include <algorithm>
#include <cmath>
namespace path_planning {
class MotionAdapter {
 public:
  static constexpr std::size_t kMaxSegments = 256;
  static constexpr std::size_t kMaxChecks = 100000;
  using Clock = std::chrono::steady_clock;
  explicit MotionAdapter(KinodynamicAstar& core)
      : core_(core), deadline_(Clock::now()+std::chrono::duration_cast<Clock::duration>(
          std::chrono::duration<double>(core.config().search_budget))) {}
  int line(const Eigen::Vector3d& start, const Eigen::Vector3d& goal) {
    if (!core_.inBounds(start) || !core_.inBounds(goal)) return KinodynamicAstar::INVALID_INPUT;
    if (!core_.mapReady()) return KinodynamicAstar::NO_MAP;
    const double intervals=std::ceil((goal-start).norm()/core_.config().collision_step);
    std::size_t n;
    if (!reserveChecks(intervals,n)) return KinodynamicAstar::INVALID_INPUT;
    for (std::size_t i=0;i<=n;++i) {
      const int status=point(start+(goal-start)*(double(i)/n));
      if (status!=KinodynamicAstar::REACH_END) return status;
    }
    return KinodynamicAstar::REACH_END;
  }
  int trajectory(const std::vector<TrajectorySegment>& segments, double elapsed,
                 const Eigen::Vector3d& goal, double max_duration) {
    if (segments.empty() || segments.size()>kMaxSegments || !std::isfinite(elapsed) || elapsed<0 ||
        !core_.inBounds(goal)) return KinodynamicAstar::INVALID_INPUT;
    double total=0;
    Eigen::Vector3d previous_position=Eigen::Vector3d::Zero(),previous_velocity=Eigen::Vector3d::Zero();
    for (std::size_t i=0;i<segments.size();++i) {
      const auto& s=segments[i];
      if (!std::isfinite(s.duration) || s.duration<=0 || !s.coefficients.allFinite() ||
          !std::isfinite(total+=s.duration) || total>max_duration) return KinodynamicAstar::INVALID_INPUT;
      if (i && ((s.coefficients.col(0)-previous_position).norm()>1e-5 ||
                (s.coefficients.col(1)-previous_velocity).norm()>1e-5)) return KinodynamicAstar::INVALID_INPUT;
      previous_position=position(s,s.duration);
      previous_velocity=velocity(s,s.duration);
      if (!previous_position.allFinite() || !previous_velocity.allFinite()) return KinodynamicAstar::INVALID_INPUT;
    }
    if (elapsed>total || (previous_position-goal).norm()>1e-6) return KinodynamicAstar::INVALID_INPUT;
    if (!core_.mapReady()) return KinodynamicAstar::NO_MAP;
    double offset=0;
    for (const auto& s:segments) {
      const double begin=std::clamp(elapsed-offset,0.0,s.duration);
      offset+=s.duration;
      if (begin==s.duration && offset<total) continue;
      const auto first=velocity(s,begin),last=velocity(s,s.duration);
      const Eigen::Vector3d middle=first+
        (s.coefficients.col(2)+3*s.coefficients.col(3)*begin)*(s.duration-begin);
      const double speed=std::max({first.norm(),middle.norm(),last.norm()});
      std::size_t n;
      if (!reserveChecks(std::ceil((s.duration-begin)*speed/core_.config().collision_step),n))
        return KinodynamicAstar::INVALID_INPUT;
      for (std::size_t i=0;i<=n;++i) {
        const int status=point(position(s,begin+(s.duration-begin)*double(i)/n));
        if (status!=KinodynamicAstar::REACH_END) return status;
      }
    }
    return KinodynamicAstar::REACH_END;
  }
 private:
  static Eigen::Vector3d position(const TrajectorySegment& s,double t) {
    return ((s.coefficients.col(3)*t+s.coefficients.col(2))*t+s.coefficients.col(1))*t+s.coefficients.col(0);
  }
  static Eigen::Vector3d velocity(const TrajectorySegment& s,double t) {
    return (3*s.coefficients.col(3)*t+2*s.coefficients.col(2))*t+s.coefficients.col(1);
  }
  bool reserveChecks(double intervals,std::size_t& n) {
    if (!std::isfinite(intervals) || intervals<0 || intervals>=kMaxChecks) return false;
    n=std::max<std::size_t>(1,static_cast<std::size_t>(intervals));
    if (n+1>kMaxChecks-checks_) return false;
    checks_+=n+1;
    return true;
  }
  int point(const Eigen::Vector3d& p) {
    if (Clock::now()>=deadline_) return KinodynamicAstar::TIMEOUT;
    if (!core_.inBounds(p)) return KinodynamicAstar::INVALID_INPUT;
    return core_.isSafe(p.x(),p.y(),p.z()) ? KinodynamicAstar::REACH_END : KinodynamicAstar::NO_PATH;
  }
  KinodynamicAstar& core_;
  Clock::time_point deadline_;
  std::size_t checks_=0;
};
}

inline void rejectOutput(msg::PlanResult& result, nav_msgs::msg::Path& path,
                         uint8_t status, const std::string& detail) {
  result.status = status;
  result.detail = detail;
  result.segments.clear();
  result.trajectory.points.clear();
  path.poses.clear();
}
template<class Fresh, class WithinBudget>
void finalizeOutput(msg::PlanResult& result, nav_msgs::msg::Path& path,
                    Fresh&& fresh, WithinBudget&& within_budget) {
  if (result.status != msg::PlanResult::REACH_END &&
      result.status != msg::PlanResult::REACH_HORIZON) return;
  if (!std::forward<Fresh>(fresh)()) {
    rejectOutput(result, path, msg::PlanResult::STALE_INPUT, "input expired during output construction");
  } else if (!std::forward<WithinBudget>(within_budget)()) {
    rejectOutput(result, path, msg::PlanResult::TIMEOUT, "output construction budget exceeded");
  }
}

class Planner : public rclcpp::Node {
 public:
  Planner() : Node("path_planning") {
    SearchConfig c;
#define PARAM(field) c.field = declare_parameter("search." #field, c.field)
    PARAM(max_tau); PARAM(init_max_tau); PARAM(max_vel); PARAM(max_acc); PARAM(w_time);

    PARAM(horizon); PARAM(lambda_heu); PARAM(resolution); PARAM(time_resolution);
    PARAM(safe_distance); PARAM(voxel_size); PARAM(collision_step); PARAM(search_budget);
    PARAM(allocate_num); PARAM(tree_period);
#undef PARAM
    c.max_vel=std::max({c.max_vel,c.max_vel_xy,c.max_vel_z});

    const int max_points = declare_parameter("search.max_cloud_points", static_cast<int>(c.max_cloud_points));
    if (max_points < 1) throw std::invalid_argument("max_cloud_points must be positive");
    c.max_cloud_points = max_points;
    auto lower = declare_parameter("search.lower", std::vector<double>{c.lower.x(),c.lower.y(),c.lower.z()});
    auto upper = declare_parameter("search.upper", std::vector<double>{c.upper.x(),c.upper.y(),c.upper.z()});
    if (lower.size()!=3 || upper.size()!=3) throw std::invalid_argument("bounds require three coordinates");
    c.lower=Eigen::Map<Vector>(lower.data()); c.upper=Eigen::Map<Vector>(upper.data());
    core_ = std::make_unique<KinodynamicAstar>(c);
    frame_=declare_parameter("planning_frame", "odom");
    rate_=declare_parameter("planning_rate",10.0); timeout_=declare_parameter("input_timeout",0.5);
    replan_interval_=declare_parameter("replan_interval",0.2);
    replan_distance_=declare_parameter("replan_distance",1.0);
    step_=declare_parameter("sample_step",0.02); stamp_clock_=declare_parameter("stamp_clock","receive");
    max_samples_=declare_parameter("output.max_samples",10000);
    max_duration_=declare_parameter("output.max_duration",60.0);
    build_budget_=declare_parameter("output.build_budget",0.1);
    body_twist_=declare_parameter("odometry_twist_in_body",true);
    blind_radius_=declare_parameter("cloud.blind_radius",0.0);
    if (declare_parameter("cloud.map_mode",std::string("dual_tree"))!="dual_tree")
      throw std::invalid_argument("cloud.map_mode must be dual_tree");
    planner_session_=std::to_string(Clock::now().time_since_epoch().count());
    blind_pose_tolerance_=declare_parameter("cloud.blind_pose_tolerance",0.05);
    blind_wait_=declare_parameter("cloud.blind_wait",0.15);
    const auto offset=declare_parameter("cloud.blind_origin_offset",std::vector<double>{0,0,0});
    if (offset.size()!=3) throw std::invalid_argument("blind_origin_offset requires three coordinates");
    blind_offset_=Eigen::Map<const Vector>(offset.data());
    if (!std::isfinite(blind_radius_) || blind_radius_<0 || !positive(blind_pose_tolerance_) ||
        !positive(blind_wait_) || blind_wait_>timeout_ ||
        !blind_offset_.allFinite()) throw std::invalid_argument("invalid blind filter parameters");
    if (frame_.empty() || !positive(rate_) || rate_>kMaxPlanningRateHz || !positive(timeout_) || !positive(step_) ||
        max_samples_<1 || max_samples_>kMaxOutputSamples || !positive(max_duration_) ||
        max_duration_>kMaxOutputDuration || !positive(build_budget_) ||
        !(replan_interval_>0) || !(replan_distance_>0) ||
        (stamp_clock_!="ros" && stamp_clock_!="receive")) throw std::invalid_argument("invalid wrapper parameters");
    auto sensor_qos=rclcpp::SensorDataQoS().keep_last(1);
    cloud_sub_=create_subscription<sensor_msgs::msg::PointCloud2>("cloud",sensor_qos,
        [this](sensor_msgs::msg::PointCloud2::ConstSharedPtr m){ cloud(*m); });
    odom_sub_=create_subscription<nav_msgs::msg::Odometry>("odom",sensor_qos,
        [this](nav_msgs::msg::Odometry::ConstSharedPtr m){ odom(*m); });
    goal_sub_=create_subscription<geometry_msgs::msg::PoseStamped>("goal",rclcpp::QoS(1),
        [this](geometry_msgs::msg::PoseStamped::ConstSharedPtr m){ goal(*m); });
    index_sub_=create_subscription<std_msgs::msg::Int64>("time_index",rclcpp::QoS(1000),
        [this](std_msgs::msg::Int64::ConstSharedPtr m){ replan_index_=m->data; });
    result_pub_=create_publisher<Result>("plan_result",rclcpp::QoS(1));
    path_pub_=create_publisher<nav_msgs::msg::Path>("kino_path",rclcpp::QoS(1));
    filtered_pub_=create_publisher<sensor_msgs::msg::PointCloud2>("filtered_cloud",sensor_qos);
    motion_service_=create_service<path_planning::srv::PlanMotion>("plan_motion",
      [this](const path_planning::srv::PlanMotion::Request::SharedPtr request,
             path_planning::srv::PlanMotion::Response::SharedPtr response) { motion(*request,*response); });
    goal_service_=create_service<path_planning::srv::SetGoal>("set_goal",
      [this](const path_planning::srv::SetGoal::Request::SharedPtr request,
             path_planning::srv::SetGoal::Response::SharedPtr response) {
        setGoal(*request,*response);
      });
    timer_=create_wall_timer(std::chrono::duration<double>(1/rate_),[this]{
      processPendingCloud();
      if (!goal_active_) { identityHeartbeat(); return; }
      if (goal_status_ || odom_status_ || cloud_status_ || !fresh(cloud_received_,cloud_stamp_) ||
          !fresh(odom_received_,odom_stamp_)) executePlan(position_,velocity_,goal_,false);
      else {
        replan(false);
        identityHeartbeat();
      }
    });
    RCLCPP_INFO(get_logger(),"Planner ready: frame=%s, replan_interval=%.3fs, replan_distance=%.3fm, clock=%s",
        frame_.c_str(),replan_interval_,replan_distance_,stamp_clock_.c_str());
  }
 private:
  static bool positive(double d) { return std::isfinite(d) && d>0; }
  static Vector vec(const geometry_msgs::msg::Point& p) { return {p.x,p.y,p.z}; }
  static Vector vec(const geometry_msgs::msg::Vector3& p) { return {p.x,p.y,p.z}; }
  bool correctFrame(const std_msgs::msg::Header& h) const { return h.frame_id==frame_; }
  static bool validStamp(const builtin_interfaces::msg::Time& stamp) {
    return stamp.sec>=0 && stamp.nanosec<kNanosecondsPerSecond;
  }
  bool fresh(Clock::time_point received, const builtin_interfaces::msg::Time& stamp) const {
    if (std::chrono::duration<double>(Clock::now()-received).count()>timeout_) return false;
    if (stamp_clock_=="receive") return true;
    const double age=(now()-rclcpp::Time(stamp)).seconds();
    return age>=-kFutureStampToleranceSeconds && age<=timeout_;
  }
  void invalidateCloud(uint8_t status, const std::string& detail) {
    pending_cloud_.reset();
    core_->clearMap(); cloud_status_=status; cloud_detail_=detail;
  }
  static bool validXyzLayout(const sensor_msgs::msg::PointCloud2& m) {
    std::array<uint32_t,kXyzFieldCount> offsets{};
    size_t coordinate=0;
    for (const auto* name:{"x","y","z"}) {
      const sensor_msgs::msg::PointField* selected=nullptr;
      for (const auto& field:m.fields) {
        if (field.name!=name) continue;
        if (selected) return false;
        selected=&field;
      }
      if (!selected || selected->datatype!=sensor_msgs::msg::PointField::FLOAT32 ||
          selected->count!=1 || uint64_t(selected->offset)+kFloat32Bytes>m.point_step) return false;
      offsets[coordinate++]=selected->offset;
    }
    for (size_t i=0;i<offsets.size();++i)
      for (size_t j=i+1;j<offsets.size();++j)
        if (uint64_t(offsets[i])<uint64_t(offsets[j])+kFloat32Bytes &&
            uint64_t(offsets[j])<uint64_t(offsets[i])+kFloat32Bytes) return false;
    return true;
  }
  void cloud(const sensor_msgs::msg::PointCloud2& m) {
    if (!correctFrame(m.header)) { invalidateCloud(Result::FRAME_MISMATCH,"cloud frame mismatch"); return; }
    if (!validStamp(m.header.stamp)) { invalidateCloud(Result::INVALID_INPUT,"invalid cloud timestamp"); return; }
    const int64_t stamp=rclcpp::Time(m.header.stamp).nanoseconds();
    if (have_cloud_stamp_ && stamp<=last_cloud_stamp_) {
      if (stamp<last_cloud_stamp_) { invalidateCloud(Result::STALE_INPUT,"cloud time moved backwards; map reset"); last_cloud_stamp_=stamp; }
      return;
    }
    last_cloud_stamp_=stamp; have_cloud_stamp_=true;
    const uint64_t count=uint64_t(m.width)*m.height;
    bool valid=count>0 && count<=core_->config().max_cloud_points && !m.is_bigendian && m.point_step>=kXyzFieldCount*kFloat32Bytes &&
        uint64_t(m.row_step)>=uint64_t(m.width)*m.point_step && m.data.size()==uint64_t(m.row_step)*m.height;
    valid=valid && validXyzLayout(m);
    if (!valid) { invalidateCloud(Result::INVALID_INPUT,"empty, oversized or malformed XYZ cloud"); return; }
    const auto received=Clock::now();
    if (!fresh(received,m.header.stamp)) { invalidateCloud(Result::STALE_INPUT,"cloud stamp outside age limit"); return; }
    if (cloud_status_==0 && !fresh(cloud_received_,cloud_stamp_)) core_->clearMap();
    pcl::PointCloud<pcl::PointXYZ> points;
    pcl::fromROSMsg(m,points);
    processPendingCloud();
    if (pending_cloud_) return;
    pending_cloud_=std::make_unique<PendingCloud>(PendingCloud{std::move(points),m.header,received});
    processPendingCloud();
  }
  void processPendingCloud() {
    if (!pending_cloud_) return;
    auto& pending=*pending_cloud_;
    if (!fresh(pending.received,pending.header.stamp)) {
      invalidateCloud(Result::STALE_INPUT,"pending cloud expired"); return;
    }
    auto& points=pending.points;
    const int64_t stamp=rclcpp::Time(pending.header.stamp).nanoseconds();
    if (blind_radius_>0) {
      const auto pose=std::min_element(blind_poses_.begin(),blind_poses_.end(),
          [stamp](const BlindPose& a,const BlindPose& b) {
            return std::abs(a.stamp-stamp)<std::abs(b.stamp-stamp);
          });
      if (odom_status_ || !fresh(odom_received_,odom_stamp_) || pose==blind_poses_.end() ||
          std::abs(pose->stamp-stamp)*1e-9>blind_pose_tolerance_) {
        if (std::chrono::duration<double>(Clock::now()-pending.received).count()>=blind_wait_)
          invalidateCloud(Result::STALE_INPUT,"blind filter timed out waiting for matched odometry");
        return;
      }
      points.erase(std::remove_if(points.begin(),points.end(),[&](const pcl::PointXYZ& p) {
        const Vector point(p.x,p.y,p.z);
        return point.allFinite() && (point-pose->origin).norm()<blind_radius_;
      }),points.end());
      if (points.empty()) {
        sensor_msgs::msg::PointCloud2 filtered; pcl::toROSMsg(points,filtered);
        filtered.header=pending.header; filtered_pub_->publish(filtered);
        invalidateCloud(Result::NO_MAP,"all cloud points removed by blind filter"); return;
      }
    }
    if (!core_->setKdtree(points)) { invalidateCloud(Result::INVALID_INPUT,"nonfinite cloud or map capacity exceeded"); return; }
    sensor_msgs::msg::PointCloud2 filtered; pcl::toROSMsg(points,filtered);
    filtered.header=pending.header; filtered_pub_->publish(filtered);
    cloud_received_=pending.received; cloud_stamp_=pending.header.stamp; cloud_status_=0;
    pending_cloud_.reset();
    replan();
  }
  void odom(const nav_msgs::msg::Odometry& m) {
    if (!correctFrame(m.header)) { odom_status_=Result::FRAME_MISMATCH; return; }
    if (!validStamp(m.header.stamp)) { odom_status_=Result::INVALID_INPUT; return; }
    const int64_t stamp=rclcpp::Time(m.header.stamp).nanoseconds();
    if (have_odom_stamp_ && stamp<=last_odom_stamp_) {
      if (stamp<last_odom_stamp_) {
        odom_status_=Result::STALE_INPUT; last_odom_stamp_=stamp; blind_poses_.clear();
        if (blind_radius_>0) invalidateCloud(Result::STALE_INPUT,"odometry time moved backwards; map reset");
      }
      return;
    }
    have_odom_stamp_=true; last_odom_stamp_=stamp;
    position_=vec(m.pose.pose.position); velocity_=vec(m.twist.twist.linear);
    const auto& q=m.pose.pose.orientation;
    Eigen::Quaterniond rotation(q.w,q.x,q.y,q.z);
    if (!position_.allFinite() || !velocity_.allFinite() || !rotation.coeffs().allFinite() ||
        std::abs(rotation.norm()-1)>kQuaternionNormTolerance || (body_twist_ && m.child_frame_id.empty())) {
      odom_status_=Result::INVALID_INPUT; return;
    }
    rotation.normalize(); if (body_twist_) velocity_=rotation*velocity_;
    odom_received_=Clock::now(); odom_stamp_=m.header.stamp;
    odom_status_=fresh(odom_received_,odom_stamp_)?0:Result::STALE_INPUT;
    if (blind_radius_>0 && !odom_status_) {
      blind_poses_.push_back({stamp,position_+rotation*blind_offset_});
      if (blind_poses_.size()>kBlindPoseHistorySize) blind_poses_.pop_front();
    }
    processPendingCloud();
    replan(false);
  }
  void fillDynamics(Result& result) const {
    const auto& config=core_->config();
    result.max_velocity_xy=std::min(config.max_vel,config.max_vel_xy);
    result.max_velocity_z=std::min(config.max_vel,config.max_vel_z);
    result.max_acceleration_xy=std::min(config.max_acc,config.max_acc_xy);
    result.max_acceleration_z=std::min(config.max_acc,config.max_acc_z);
  }
  std::string searchDetail(int status,const Vector& start,const Vector& velocity,const Vector& goal) {
    const auto& c=core_->config();
    if (status==Result::REACH_END) return "goal reached by trajectory";
    if (status==Result::REACH_HORIZON) return "partial horizon; endpoint is not goal";
    if (status==Result::INVALID_INPUT) {
      if (!start.allFinite() || !velocity.allFinite() || !goal.allFinite()) return "nonfinite search boundary state";
      const auto bounds_detail=[&](const char* label,const Vector& point) {
        return std::string(label)+" outside search bounds: position=["+
            std::to_string(point.x())+","+std::to_string(point.y())+","+std::to_string(point.z())+
            "], lower=["+std::to_string(c.lower.x())+","+std::to_string(c.lower.y())+","+std::to_string(c.lower.z())+
            "], upper=["+std::to_string(c.upper.x())+","+std::to_string(c.upper.y())+","+std::to_string(c.upper.z())+"]";
      };
      if (!core_->inBounds(start)) return bounds_detail("start",start);
      if (!core_->inBounds(goal)) return bounds_detail("goal",goal);
      return "invalid search boundary state or unrepresentable braking duration";
    }
    if (status==Result::NO_MAP) return "obstacle map unavailable";
    if (status==Result::TIMEOUT) return "search time budget exhausted";
    if (status==Result::NODE_LIMIT) return "search node capacity exhausted";
    if (!core_->isSafe(start.x(),start.y(),start.z())) return "start violates obstacle clearance";
    if (!core_->isSafe(goal.x(),goal.y(),goal.z())) return "goal violates obstacle clearance";
    if (velocity.head<2>().norm()>c.max_vel_xy || std::abs(velocity.z())>c.max_vel_z ||
        velocity.cwiseAbs().maxCoeff()>c.max_vel)
      return "no feasible trajectory from overspeed start: bounded braking or continuation blocked";
    return "no feasible trajectory: search frontier exhausted";
  }
  void identityHeartbeat() {
    const auto current=Clock::now();
    if (std::chrono::duration<double>(current-last_result_publication_).count()<kIdentityHeartbeatSeconds) return;
    Result result; result.header.stamp=now(); result.header.frame_id=frame_;
    fillDynamics(result);
    result.planner_session=planner_session_; result.sequence=++sequence_; result.goal_revision=goal_revision_;
    result.status=Result::WAITING_FOR_INPUT; result.detail="planner identity heartbeat";
    result.cloud_stamp=cloud_stamp_; result.odometry_stamp=odom_stamp_; result.map_points=core_->mapPointCount();
    result_pub_->publish(result); last_result_publication_=current;
  }
  void clearGoalHistory() {
    core_->reset(); standalone_path_.clear(); successful_curve_valid_=false; curve_recovery_pending_=false; first_plan_=true; replan_flag_=false;
    replan_index_=-1; last_replan_index_=-1; last_replan_time_=Clock::time_point{};
    last_replan_position_=position_;
  }
  void setGoal(const path_planning::srv::SetGoal::Request& request,
               path_planning::srv::SetGoal::Response& response) {
    response.planner_session=planner_session_; response.goal_revision=goal_revision_;
    if (!request.active) {
      if (goal_active_ && !request.target_id.empty() && request.target_id!=target_id_) {
        response.detail="active target identity mismatch"; return;
      }
      if (goal_active_) {
        ++goal_revision_; clearGoalHistory(); goal_active_=false; target_id_.clear();
        goal_status_=Result::WAITING_FOR_INPUT;
      }
      response.accepted=true; response.goal_revision=goal_revision_; response.detail="target inactive";
      return;
    }
    const Vector next=vec(request.goal);
    if (!correctFrame(request.header) || !next.allFinite() || !core_->inBounds(next)) {
      response.detail="invalid target frame or position"; return;
    }
    if (request.target_id.empty()) { response.detail="active target requires target_id"; return; }
    if (goal_active_ && request.target_id==target_id_ && next==goal_) {
      response.accepted=true; response.detail="target unchanged"; return;
    }
    ++goal_revision_; clearGoalHistory(); goal_=next; target_id_=request.target_id;
    goal_active_=true; goal_status_=0;
    response.accepted=true; response.goal_revision=goal_revision_; response.detail="target accepted";
    if (!odom_status_) plan();
  }
  void goal(const geometry_msgs::msg::PoseStamped& m) {
    path_planning::srv::SetGoal::Request request;
    request.header=m.header; request.goal=m.pose.position; request.active=true;
    request.target_id="manual:"+planner_session_+":"+std::to_string(++manual_goal_sequence_);
    path_planning::srv::SetGoal::Response response;
    setGoal(request,response);
    if (!response.accepted) RCLCPP_WARN(get_logger(),"Manual goal rejected: %s",response.detail.c_str());
  }
  void replan(bool check_path=true) {
    if (!goal_active_ || goal_status_ || odom_status_ || cloud_status_ ||
        (successful_curve_valid_ && (goal_-position_).norm()<kNearGoalDistanceMeters) ||
        !fresh(cloud_received_,cloud_stamp_) || !fresh(odom_received_,odom_stamp_)) return;
    if (curve_recovery_pending_) { plan(true); return; }
    if (std::chrono::duration<double>(Clock::now()-last_replan_time_).count()>=replan_interval_ ||
        (position_-last_replan_position_).norm()>=replan_distance_) {
      plan(true);
      return;
    }
    if (!check_path || standalone_path_.empty()) return;
    for (size_t i=0;i<standalone_path_.size();i+=40) {
      if (replan_flag_) break;
      const auto& p=standalone_path_[i];
      replan_flag_=!core_->isSafe(p.x(),p.y(),p.z());
    }
    if (replan_flag_) plan();
  }
  void plan(bool refresh=false) {
    Vector start=position_,velocity=velocity_;
    bool from_cached_path=false;
    if (!refresh && !first_plan_ && replan_index_!=-1) {
      if (std::abs(static_cast<long double>(last_replan_index_)-replan_index_)<10) return;
      const auto n=static_cast<int64_t>(standalone_path_.size());
      if (replan_index_<0 || replan_index_>=n) return;
      int64_t used=replan_index_+3;
      last_replan_index_=replan_index_;
      if (used>n-4) used=replan_index_;
      start=standalone_path_[used];
      if (used+2>=n) {
        if (used<2) return;
        velocity=(standalone_path_[used]-standalone_path_[used-2])/0.02;
      } else if (used<2) {
        velocity=(standalone_path_[used+2]-standalone_path_[used])/0.02;
      } else {
        velocity=(standalone_path_[used+2]-standalone_path_[used-2])/0.04;
      }
      from_cached_path=true;
    }
    curve_recovery_pending_=false;
    last_replan_time_=Clock::now(); last_replan_position_=position_;
    const auto result=executePlan(start,velocity,goal_,false,from_cached_path);
    if (result.status==Result::REACH_END || result.status==Result::REACH_HORIZON) {
      standalone_path_=core_->getKinoTraj(0.01);
      first_plan_=false; replan_flag_=false;
    }
  }
  Result executePlan(const Vector& start,const Vector& velocity,const Vector& goal,bool requested,
                     bool preserve_initial_acceleration=false) {
    processPendingCloud();
    Result result; result.header.stamp=now(); result.header.frame_id=frame_;
    fillDynamics(result);
    result.sequence=++sequence_; result.goal_revision=goal_revision_;
    result.planner_session=planner_session_; result.target_id=requested?std::string{}:target_id_;
    result.cloud_stamp=cloud_stamp_; result.odometry_stamp=odom_stamp_;
    if (!requested && goal_status_) { result.status=goal_status_; result.detail="goal unavailable or invalid"; }
    else if (!requested && odom_status_) { result.status=odom_status_; result.detail="odometry unavailable or invalid"; }
    else if (cloud_status_) { result.status=cloud_status_; result.detail=cloud_detail_; }
    else if (!fresh(cloud_received_,cloud_stamp_) || (!requested && !fresh(odom_received_,odom_stamp_))) {
      result.status=Result::STALE_INPUT; result.detail="cloud or odometry expired";
      if (!fresh(cloud_received_,cloud_stamp_)) invalidateCloud(Result::STALE_INPUT,"cloud expired; map reset");
    } else {
      const auto began=Clock::now();
      result.status=core_->search(start,velocity,Vector::Zero(),goal,Vector::Zero(),preserve_initial_acceleration);
      if (preserve_initial_acceleration && result.status==Result::NO_PATH)
        result.status=core_->search(start,velocity,Vector::Zero(),goal,Vector::Zero(),false);
      result.planning_ms=std::chrono::duration<double,std::milli>(Clock::now()-began).count();
      result.detail=searchDetail(result.status,start,velocity,goal);
      if (!fresh(cloud_received_,cloud_stamp_) || (!requested && !fresh(odom_received_,odom_stamp_))) {
        result.status=Result::STALE_INPUT; result.detail="input expired during search";
      }
    }
    nav_msgs::msg::Path path; path.header=result.header;
    result.trajectory.header=result.header; result.trajectory.joint_names={"position"};
    const auto output_began=Clock::now();
    const auto within_budget=[&] {
      return std::chrono::duration<double>(Clock::now()-output_began).count()<=build_budget_;
    };
    if (result.status==Result::REACH_END || result.status==Result::REACH_HORIZON) {
      try {
        const auto segments=core_->getSegments();
        double duration=0;
        for (const auto& s:segments) {
          duration+=s.duration;
          if (!positive(s.duration) || !std::isfinite(duration) || duration>max_duration_)
            throw std::length_error("output trajectory duration limit exceeded");
        }
        for (const auto& s:segments) {
          path_planning::msg::PolynomialSegment msg; msg.duration=s.duration;
          for (int i=0;i<4;++i) { msg.x[i]=s.coefficients(0,i); msg.y[i]=s.coefficients(1,i); msg.z[i]=s.coefficients(2,i); }
          result.segments.push_back(msg);
        }
        const auto samples=core_->sampleTrajectory(step_,static_cast<size_t>(max_samples_));
        result.trajectory.points.reserve(samples.size()); path.poses.reserve(samples.size());
        for (const auto& s:samples) {
          if (!within_budget()) {
            path_planning::rejectOutput(result,path,Result::TIMEOUT,"output construction budget exceeded");
            break;
          }
          trajectory_msgs::msg::MultiDOFJointTrajectoryPoint point;
          geometry_msgs::msg::Transform p; p.rotation.w=1;
          p.translation.x=s.position.x(); p.translation.y=s.position.y(); p.translation.z=s.position.z();
          geometry_msgs::msg::Twist v,a;
          v.linear.x=s.velocity.x(); v.linear.y=s.velocity.y(); v.linear.z=s.velocity.z();
          a.linear.x=s.acceleration.x(); a.linear.y=s.acceleration.y(); a.linear.z=s.acceleration.z();
          point.transforms.push_back(p); point.velocities.push_back(v); point.accelerations.push_back(a);
          point.time_from_start=rclcpp::Duration::from_seconds(s.time); result.trajectory.points.push_back(point);
          geometry_msgs::msg::PoseStamped pose; pose.header=result.header;
          pose.header.stamp=rclcpp::Time(result.header.stamp)+rclcpp::Duration::from_seconds(s.time);
          pose.pose.position.x=s.position.x(); pose.pose.position.y=s.position.y(); pose.pose.position.z=s.position.z();
          pose.pose.orientation.w=1; path.poses.push_back(pose);
        }
      } catch (const std::exception& e) {
        path_planning::rejectOutput(result,path,Result::INVALID_INPUT,e.what());
      }
    }
    result.map_points=core_->mapPointCount();
    path_planning::finalizeOutput(result,path,[&] {
      return fresh(cloud_received_,cloud_stamp_) && (requested || fresh(odom_received_,odom_stamp_));
    },within_budget);
    if (!requested) {
      const bool success=result.status==Result::REACH_END || result.status==Result::REACH_HORIZON;
      if (successful_curve_valid_ && !success) curve_recovery_pending_=true;
      successful_curve_valid_=success;
      last_result_publication_=Clock::now();
      result_pub_->publish(result);
      if (!path.poses.empty()) path_pub_->publish(path);
    }
    return result;
  }
  void motion(const path_planning::srv::PlanMotion::Request& request,
              path_planning::srv::PlanMotion::Response& response) {
    using Service=path_planning::srv::PlanMotion;
    processPendingCloud(); response.planner_session=planner_session_;
    auto& result=response.result;
    fillDynamics(result);
    result.header.stamp=now(); result.header.frame_id=frame_; result.sequence=++sequence_;
    result.planner_session=planner_session_; result.goal_revision=goal_revision_;
    result.cloud_stamp=cloud_stamp_; result.map_points=core_->mapPointCount();
    const auto received=Clock::now();
    const double request_age=validStamp(request.header.stamp) ?
      (now()-rclcpp::Time(request.header.stamp)).seconds() : std::numeric_limits<double>::infinity();
    const auto request_fresh=[&] {
      return request_age>=-kFutureStampToleranceSeconds &&
        request_age+std::chrono::duration<double>(Clock::now()-received).count()<=timeout_;
    };
    if (!correctFrame(request.header)) { result.status=Result::FRAME_MISMATCH; result.detail="request frame mismatch"; }
    else if (!validStamp(request.header.stamp) || !vec(request.start).allFinite() ||
             !vec(request.velocity).allFinite() || !vec(request.goal).allFinite() ||
             request.mode<Service::Request::PLAN || request.mode>Service::Request::VALIDATE_TRAJECTORY) {
      result.status=Result::INVALID_INPUT; result.detail="invalid request state, mode or timestamp";
    } else if (!request_fresh()) { result.status=Result::STALE_INPUT; result.detail="request expired"; }
    else if (cloud_status_) { result.status=cloud_status_; result.detail=cloud_detail_; }
    else if (!fresh(cloud_received_,cloud_stamp_)) {
      invalidateCloud(Result::STALE_INPUT,"cloud expired; map reset");
      result.status=Result::STALE_INPUT; result.detail=cloud_detail_;
    } else {
      const auto began=Clock::now();
      if (request.mode==Service::Request::PLAN) {
        result=executePlan(vec(request.start),vec(request.velocity),vec(request.goal),true);
      } else if (request.mode==Service::Request::CHECK_LINE) {
        result.status=path_planning::MotionAdapter(*core_).line(vec(request.start),vec(request.goal));
      } else if (request.segments.size()>path_planning::MotionAdapter::kMaxSegments) {
        result.status=Result::INVALID_INPUT;
      } else {
        std::vector<TrajectorySegment> segments;
        segments.reserve(request.segments.size());
        for (const auto& wire:request.segments) {
          TrajectorySegment segment; segment.duration=wire.duration;
          for (int i=0;i<4;++i) {
            segment.coefficients(0,i)=wire.x[i]; segment.coefficients(1,i)=wire.y[i]; segment.coefficients(2,i)=wire.z[i];
          }
          segments.push_back(segment);
        }
        result.status=path_planning::MotionAdapter(*core_).trajectory(
          segments,request.trajectory_elapsed,vec(request.goal),max_duration_);
      }
      result.planning_ms=std::chrono::duration<double,std::milli>(Clock::now()-began).count();
      if (request.mode!=Service::Request::PLAN)
        result.detail=result.status==Result::REACH_END ? "sampled current-map geometry clear" : "geometry blocked or unavailable";
      if (!fresh(cloud_received_,cloud_stamp_) || !request_fresh()) {
        result.status=Result::STALE_INPUT; result.detail="input expired during service";
        result.segments.clear(); result.trajectory.points.clear();
      }
    }
    result.map_points=core_->mapPointCount();
  }

PLANNER_KEYS = {

def load_config(path):
    try:
        with open(path, encoding='utf-8') as stream:
            data = yaml.safe_load(stream)
        if not isinstance(data, dict):
            raise ValueError('expected a mapping of node names')
        return data
    except (OSError, ValueError, yaml.YAMLError) as error:
        print(f'[planner config] Cannot read {path}: {error}; using defaults.', file=sys.stderr)
        return {}
def flatten(parameters, prefix=''):
    result = {}
    for key, value in parameters.items():
        name = f'{prefix}{key}'
        if isinstance(value, dict):
            result.update(flatten(value, name + '.'))
        else:
            result[name] = value
    return result
def node_parameters(data, names, defaults=None):
    result = dict(defaults or {})
    for name in names:
        if name not in data:
            continue
        section = data[name]
        parameters = section.get('ros__parameters') if isinstance(section, dict) else None
        if not isinstance(parameters, dict):
            print(f'[planner config] Invalid {name}.ros__parameters; using defaults for this section.',
                  file=sys.stderr)
            continue
        result.update(flatten(parameters))
    return result
def planner_overrides(context, parameters):
    result = dict(parameters)
    arguments = context.launch_configurations
    for alias, name in ALIASES.items():
        if arguments.get(alias) and arguments.get(name):
            raise ValueError(f'Supply only one of {alias} and {name}')
    for key, raw in arguments.items():
        name = ALIASES.get(key, key)
        if not raw or not (name in PLANNER_KEYS or name.startswith(('search.', 'cloud.', 'output.'))):
            continue
        value = yaml.safe_load(raw)
        if name in ('planning_frame', 'stamp_clock', 'cloud.map_mode'):
            value = raw
        elif name in ('odometry_twist_in_body',):
            if not isinstance(value, bool):
                raise ValueError(f'{key} must be true or false')
        elif name in INTEGER_KEYS:
            if isinstance(value, bool) or not isinstance(value, int):
                raise ValueError(f'{key} must be an integer')
        elif isinstance(value, list):
            value = [float(item) for item in value]
        else:
            value = float(value)
        result[name] = value
    for scalar in ('search.max_vel', 'search.max_acc'):
        if arguments.get(scalar):
            for suffix in ('_xy', '_z'):
                if not arguments.get(scalar+suffix):
                    result[scalar+suffix] = result[scalar]
    return result
def launch_value(context, name, configured):
    return context.launch_configurations.get(name) or configured

      result.status = Result::WAITING_FOR_INPUT;
      result.detail = reason_;
      result.plan_sequence = 0;
      result_pub_->publish(result);
    });
  }
private:
  void connect_native(const std::string &selected) {
    native_namespace_ = selected;
    planner_session_.clear();
    native_time_ = {};
    last_native_sequence_ = 0;
    motion_limits_ = {};
    const auto generation = ++native_generation_;
    goal_client_ = create_client<Goal>(selected + "/set_goal");
    motion_client_ = create_client<Motion>(selected + "/plan_motion");
    native_sub_ = create_subscription<NativeResult>(selected + "/plan_result", rclcpp::QoS(10),
      [this, generation](NativeResult::SharedPtr message) {
        if (generation == native_generation_) { native_result(*message); }
      });
    if (preview_) {
      cloud_sub_ = create_subscription<Cloud>(selected + "/filtered_cloud", rclcpp::SensorDataQoS(),
        [this, generation](Cloud::SharedPtr message) {
          if (generation == native_generation_) { preview(*message); }
        });
    }
  }
  std::array<double, 3> local_origin_offset() const {
    if (mode_ == "identity" || !status_) { return {0., 0., 0.}; }
    return {status_->lever_arm_x_m, status_->lever_arm_y_m, status_->lever_arm_z_m};
  }
  bool coordinates() {
    if (!state_ || age(state_time_) > timeout_ || !finite(state_->pose.pose.position)) {
      reason_ = "FCU odometry unavailable or stale";
      return false;
    }
    if (mode_ == "identity") {
      if (state_->header.frame_id != frame_) { reason_ = "odometry frame mismatch"; return false; }
      return true;
    }
    if (state_->header.frame_id != "map" || !status_ ||
        age(status_time_) > kCoordinateStatusTimeoutSeconds ||
        !status_->service_available || !status_->odin_available || status_->interface_version != "2.0" ||
        status_->odin_session_id.empty() || status_->correction_valid ||
        !status_->final_sample_available || status_->final_sample_correction_valid ||
        status_->final_sample_revision != status_->revision ||
        status_->final_sample_odin_session_id != status_->odin_session_id ||
        status_->final_sample_reference_mode != "local_identity_origin") {
      reason_ = "extnav local-identity coordinates unavailable (tag correction unsupported)";
      return false;
    }
    for (double value : local_origin_offset()) {
      if (!std::isfinite(value)) { reason_ = "invalid extnav translation"; return false; }
    }
    return true;
  }
  bool ready() {
    if (!coordinates()) { return false; }
    if (!goal_client_->service_is_ready() || !motion_client_->service_is_ready()) {
      reason_ = "native planner interfaces unavailable";
      return false;
    }
    if (age(native_time_) > 3.) {
      reason_ = "native planner identity heartbeat unavailable";
      return false;
    }
    reason_ = "native planner connected; map validity is reported in each planning result";
    return true;
  }
  Result identity(const Request &request) {
    Result result;
    result.header.stamp = now();
    result.header.frame_id = "map";
    result.controller_session = request.controller_session;
    result.task_revision = request.task_revision;
    result.waypoint_index = request.waypoint_index;
    result.request_id = request.request_id;
    result.mode = request.mode;
    result.bridge_session = session_;
    result.planner_session = planner_session_;
    result.coordinate_revision = coordinate_revision_;
    result.plan_sequence = ++result_sequence_;
    result.max_velocity_xy = motion_limits_[0];
    result.max_velocity_z = motion_limits_[1];
    result.max_acceleration_xy = motion_limits_[2];
    result.max_acceleration_z = motion_limits_[3];
    return result;
  }
  void remember_limits(const NativeResult &message) {
    motion_limits_ = {message.max_velocity_xy, message.max_velocity_z,
      message.max_acceleration_xy, message.max_acceleration_z};
  }
  void fail(const Request &request, uint8_t status, const std::string &detail) {
    Result result = identity(request);
    result.plan_sequence = 0;
    result.ready = ready();
    result.status = status;
    result.detail = detail;
    result_pub_->publish(result);
  }
  static std::string target_id(const Request &request) {
    return request.controller_session + ":" + std::to_string(request.task_revision) + ":" +
      std::to_string(request.waypoint_index) + ":" + std::to_string(request.request_id);
  }
  void cancel_goal() {
    if (!active_) { return; }
    auto goal = std::make_shared<Goal::Request>();
    goal->header.stamp = now();
    goal->header.frame_id = frame_;
    goal->target_id = target_id(*active_);
    goal->active = false;
    if (goal_client_->service_is_ready()) { goal_client_->async_send_request(goal); }
    active_.reset();
    goal_confirmed_ = false;
    pending_native_.reset();
    ++goal_generation_;
  }
  void request(const Request &message) {
    if (message.mode == Request::CANCEL) {
      auto goal = std::make_shared<Goal::Request>();
      goal->header.stamp = now(); goal->header.frame_id = frame_;
      goal->target_id = target_id(message); goal->active = false;
      if (goal_client_->service_is_ready()) { goal_client_->async_send_request(goal); }
      if (active_ && target_id(message) == target_id(*active_)) {
        active_.reset(); goal_confirmed_ = false; pending_native_.reset(); ++goal_generation_;
      }
      return;
    }
    if (!message.request_id) { return; }
    if (message.controller_session.empty() || message.waypoint_index == 0 ||
        message.header.frame_id != "map" || !finite(message.start) || !finite(message.goal)) {
      fail(message, Result::INVALID_INPUT, "invalid target identity or coordinates");
      return;
    }
    if (!ready()) { fail(message, Result::WAITING_FOR_INPUT, reason_); return; }
    const auto t = local_origin_offset();
    if (message.mode == Request::PLAN) {
      active_ = message;
      goal_confirmed_ = false;
      pending_native_.reset();
      const uint64_t generation = ++goal_generation_;
      const std::string id = target_id(message);
      auto goal = std::make_shared<Goal::Request>();
      goal->header.stamp = now();
      goal->header.frame_id = frame_;
      goal->target_id = id;
      goal->active = true;
      goal->goal = message.goal;
      goal->goal.x -= t[0]; goal->goal.y -= t[1]; goal->goal.z -= t[2];
      goal_client_->async_send_request(goal, [this, message, id, generation](rclcpp::Client<Goal>::SharedFuture future) {
        if (!active_ || target_id(*active_) != id || generation != goal_generation_) { return; }
        const auto response = future.get();
        planner_session_ = response->planner_session;
        if (!response->accepted) { fail(message, Result::INVALID_INPUT, response->detail); return; }
        active_goal_revision_ = response->goal_revision;
        goal_confirmed_ = true;
        if (pending_native_) {
          const auto result = *pending_native_;
          pending_native_.reset();
          native_result(result);
        }
      });
      return;
    }
    if (message.mode != Request::CHECK_LINE) {
      fail(message, Result::INVALID_INPUT, "only native goals and straight-line checks are supported");
      return;
    }
    auto query = std::make_shared<Motion::Request>();
    query->header.stamp = now(); query->header.frame_id = frame_;
    query->mode = Motion::Request::CHECK_LINE;
    query->start = message.start; query->goal = message.goal;
    query->start.x -= t[0]; query->start.y -= t[1]; query->start.z -= t[2];
    query->goal.x -= t[0]; query->goal.y -= t[1]; query->goal.z -= t[2];
    const uint64_t revision = coordinate_revision_;
    motion_client_->async_send_request(query,
      [this, message, revision](rclcpp::Client<Motion>::SharedFuture future) {
        if (revision != coordinate_revision_) { return; }
        const auto response = future.get();
        planner_session_ = response->planner_session;
        remember_limits(response->result);
        Result result = identity(message);
        result.header.stamp = response->result.header.stamp;
        result.ready = ready();
        result.status = response->result.status;
        result.detail = response->result.detail;
        result_pub_->publish(result);
      });
  }
  void native_result(const NativeResult &message) {
    if (message.planner_session.empty() || message.header.frame_id != frame_) { return; }
    native_time_ = Steady::now();
    remember_limits(message);
    if (message.planner_session != planner_session_) {
      if (!planner_session_.empty()) {
        goal_confirmed_ = false;
        pending_native_.reset();
        ++goal_generation_;
      }
      planner_session_ = message.planner_session;
      last_native_sequence_ = 0;
    }
    if (!active_) { return; }
    if (goal_confirmed_ && !message.target_id.empty() && message.goal_revision > active_goal_revision_) {
      Result revoked = identity(*active_);
      revoked.header.stamp = message.header.stamp;
      revoked.ready = ready();
      revoked.status = Result::NO_PATH;
      revoked.detail = "native target replaced; current waypoint reference revoked";
      result_pub_->publish(revoked);
      return;
    }
    if (message.target_id != target_id(*active_)) { return; }
    if (!goal_confirmed_) { pending_native_ = message; return; }
    if (message.goal_revision != active_goal_revision_ || message.sequence <= last_native_sequence_) { return; }
    last_native_sequence_ = message.sequence;
    Result result = identity(*active_);
    result.header.stamp = message.header.stamp;
    result.ready = ready();
    result.status = message.status;
    result.detail = message.detail;
    if (result.ready && (message.status == NativeResult::REACH_END ||
        message.status == NativeResult::REACH_HORIZON)) {
      const auto t = local_origin_offset();
      for (const auto &segment : message.segments) {
        guided_interfaces::msg::PolynomialSegment output;
        output.duration = segment.duration;
        output.x = segment.x; output.y = segment.y; output.z = segment.z;
        output.x[0] += t[0]; output.y[0] += t[1]; output.z[0] += t[2];
        result.segments.push_back(output);
      }
    }
    result_pub_->publish(result);
  }
  void preview(const Cloud &message) {
    constexpr size_t max_points = 10000;
    if (age(preview_time_) < .5 || !coordinates() || message.header.frame_id != frame_ ||
        preview_pub_->get_subscription_count() == 0 || message.is_bigendian || !message.width ||
        !message.height || !message.point_step || message.data.size() < size_t(message.row_step) * message.height) {
      return;
    }
    std::array<uint32_t, 3> offsets{};
    size_t index = 0;
    for (const std::string name : {"x", "y", "z"}) {
      bool found = false;
      for (const auto &field : message.fields) {
        if (field.name == name && field.datatype == sensor_msgs::msg::PointField::FLOAT32 &&
            field.count == 1 && field.offset + sizeof(float) <= message.point_step) {
          offsets[index] = field.offset;
          found = true;
          break;
        }
      }
      if (!found) { return; }
      ++index;
    }
    const size_t total = size_t(message.width) * message.height;
    const size_t count = std::min(total, max_points);
    Cloud output;
    output.header.stamp = now(); output.header.frame_id = preview_frame_;
    output.height = 1; output.width = count; output.point_step = 12; output.row_step = count * 12;
    output.is_dense = true;
    for (size_t axis = 0; axis < 3; ++axis) {
      sensor_msgs::msg::PointField field;
      field.name = axis == 0 ? "x" : axis == 1 ? "y" : "z";
      field.offset = axis * sizeof(float); field.count = 1;
      field.datatype = sensor_msgs::msg::PointField::FLOAT32;
      output.fields.push_back(field);
    }
    output.data.resize(output.row_step);
    const auto t = local_origin_offset();
    for (size_t point = 0; point < count; ++point) {
      const size_t source = point * total / count;
      const size_t base = source / message.width * message.row_step + source % message.width * message.point_step;
      for (size_t axis = 0; axis < 3; ++axis) {
        float value;
        std::memcpy(&value, message.data.data() + base + offsets[axis], sizeof(value));
        value += t[axis];
        std::memcpy(output.data.data() + point * 12 + axis * sizeof(float), &value, sizeof(value));
      }
    }
    preview_pub_->publish(output);
    preview_time_ = Steady::now();
  }
  std::string mode_, frame_, session_, planner_session_, reason_, native_namespace_;
  std::string preview_frame_{"map"};
  rclcpp::Subscription<guided_interfaces::msg::ControlStatus>::SharedPtr control_status_sub_;
  double timeout_{};
  std::array<double, 4> motion_limits_{};
  bool preview_{}, goal_confirmed_{};
  uint64_t coordinate_revision_{1}, result_sequence_{0}, last_native_sequence_{0};
  uint64_t goal_generation_{0}, active_goal_revision_{0};
  std::optional<Request> active_;
  std::optional<NativeResult> pending_native_;
  std::optional<Odom> state_;
  std::optional<Status> status_;
  std::optional<guided_interfaces::msg::ControlStatus> control_status_;
  uint64_t native_generation_{0};
  OnSetParametersCallbackHandle::SharedPtr parameter_callback_;
  Steady::time_point state_time_{}, status_time_{}, preview_time_{}, native_time_{}, control_time_{};
  rclcpp::Publisher<Result>::SharedPtr result_pub_;
  rclcpp::Publisher<Odom>::SharedPtr odom_pub_;
  rclcpp::Publisher<Cloud>::SharedPtr preview_pub_;
  rclcpp::Subscription<Request>::SharedPtr request_sub_;
  rclcpp::Subscription<NativeResult>::SharedPtr native_sub_;
  rclcpp::Subscription<Odom>::SharedPtr odom_sub_;
  rclcpp::Subscription<Status>::SharedPtr status_sub_;
  rclcpp::Subscription<Cloud>::SharedPtr cloud_sub_;
  rclcpp::Client<Goal>::SharedPtr goal_client_;
  rclcpp::Client<Motion>::SharedPtr motion_client_;
  rclcpp::TimerBase::SharedPtr heartbeat_;
};
int main(int argc, char **argv) {
  rclcpp::init(argc, argv);
  rclcpp::spin(std::make_shared<Bridge>());
  rclcpp::shutdown();
  return 0;
}
from pathlib import Path
import runpy
import yaml
from ament_index_python.packages import get_package_share_directory
from launch import LaunchDescription
from launch.actions import DeclareLaunchArgument, OpaqueFunction, Shutdown
from launch.substitutions import LaunchConfiguration
from launch_ros.actions import Node
def _nodes(context):
    helper = runpy.run_path(str(Path(get_package_share_directory('path_planning')) /
                                'launch/parameter_config.py'))
    data = helper['load_config'](LaunchConfiguration('config').perform(context))
    parameters = helper['node_parameters'](
        data, ('/**', 'path_planning', '/path_planning'), {})
    topics = helper['node_parameters'](data, ('odin_launch',), {
        'cloud': '/odin1/cloud_slam_down',
    })
    bridge = helper['node_parameters'](data, ('avoidance_bridge', '/avoidance_bridge'), {
        'coordinate_mode': 'extnav',
        'odometry_topic': '/mavros/local_position/odom', 'preview_enabled': False,
    })
    bridge['coordinate_mode'] = helper['launch_value'](context, 'coordinate_mode', bridge['coordinate_mode'])
    cloud = helper['launch_value'](context, 'cloud', topics['cloud'])
    bridge['odometry_topic'] = helper['launch_value'](context, 'odom', bridge['odometry_topic'])
    if context.launch_configurations.get('preview'):
        raw = context.launch_configurations['preview'].lower()
        if raw not in ('true', 'false'):
            raise ValueError('preview must be true or false')
        bridge['preview_enabled'] = raw == 'true'
    if context.launch_configurations.get('coordinate_mode') == 'identity':
        parameters['planning_frame'] = 'map'
    parameters = helper['planner_overrides'](context, parameters)
    bridge.setdefault('input_timeout', parameters.get('input_timeout', 0.5))
    bridge['planning_frame'] = parameters.get('planning_frame', 'odom')
    for key, raw in context.launch_configurations.items():
        if key.startswith('bridge.') and raw:
            name = key.removeprefix('bridge.')
            value = yaml.safe_load(raw)
            if isinstance(bridge.get(name), float) and isinstance(value, (int, float)):
                value = float(value)
            bridge[name] = value
    return [
        Node(package='path_planning', executable='path_planning_node', namespace='avoidance',
             name='path_planning', output='screen', parameters=[parameters],
             on_exit=Shutdown(reason='Native planner exited'),
             remappings=[('cloud', cloud), ('odom', '/avoidance/odom'),
                         ('goal', '/avoidance/goal')]),
        Node(package='avoidance_bridge', executable='avoidance_bridge_node', output='screen',
             on_exit=Shutdown(reason='Avoidance bridge exited'),
             parameters=[bridge]),
        Node(package='avoidance_bridge', executable='avoidance_backend_manager.py', output='screen',
             parameters=[{
                 'planning_frame': bridge['planning_frame'], 'cloud_topic': cloud,
                 'odometry_twist_in_body': parameters.get('odometry_twist_in_body', True),
                 'search_lower': parameters.get('search.lower', [-20., -20., .2]),
                 'search_upper': parameters.get('search.upper', [20., 20., 1.8]),
             }]),
    ]
def generate_launch_description():
    config = str(Path(get_package_share_directory('avoidance_bridge')) / 'config/planner.yaml')
    arguments = [DeclareLaunchArgument('config', default_value=config)]
    arguments.extend(DeclareLaunchArgument(name, default_value='') for name in (
        'coordinate_mode', 'cloud', 'odom', 'preview', 'blind_radius', 'map_mode',
        'planning_frame', 'stamp_clock', 'search_lower',
    ))
    return LaunchDescription([*arguments, OpaqueFunction(function=_nodes)])
